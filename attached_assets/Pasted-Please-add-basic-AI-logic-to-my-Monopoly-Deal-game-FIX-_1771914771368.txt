runAiTurn: () => {
  const state = get()
  const aiPlayerId = state.currentTurnPlayerId
  if (aiPlayerId === state.currentPlayerId) return // not an AI turn

  const aiPlayer = state.players.find((p) => p.id === aiPlayerId)
  if (!aiPlayer) return

  let aiHand = [...(state.aiHands[aiPlayerId] ?? [])]
  let actionsLeft = 3
  const actions: string[] = []

  // Draw 2 cards for AI
  const drawn = state.deck.slice(0, 2)
  aiHand = [...aiHand, ...drawn]

  // Play up to 3 cards - prioritise properties, then bank money
  const toPlay: GameCard[] = []
  const toBank: GameCard[] = []
  const remaining: GameCard[] = []

  for (const card of aiHand) {
    if (actionsLeft <= 0) {
      remaining.push(card)
      continue
    }
    if (isPropertyCard(card) || isPropertyWildcard(card)) {
      toPlay.push(card)
      actionsLeft--
    } else if (isMoneyCard(card) && actionsLeft > 0) {
      toBank.push(card)
      actionsLeft--
    } else {
      remaining.push(card)
    }
  }

  // Apply property cards to AI player's sets
  set((s) => {
    const playerIdx = s.players.findIndex((p) => p.id === aiPlayerId)
    
    for (const card of toPlay) {
      if (isPropertyCard(card)) {
        s.players[playerIdx].propertySets = addCardToPropertySets(
          s.players[playerIdx].propertySets,
          card
        )
        actions.push(`played ${card.name}`)
      } else if (isPropertyWildcard(card)) {
        const colors = Array.isArray(card.colors) 
          ? card.colors.filter((c): c is CardColor => 
              ["darkblue","lightblue","brown","green","utility","red","yellow","orange","pink","railroad","purple"].includes(c)
            )
          : []
        if (colors.length > 0) {
          const matchingSet = s.players[playerIdx].propertySets.find(
            (ps) => colors.includes(ps.color)
          )
          const chosenColor = matchingSet ? matchingSet.color : colors[0]
          const defaults = DEFAULT_SET_PROPS[chosenColor] ?? { rentTable: [], setRequired: 2 }
          const existingSet = s.players[playerIdx].propertySets.find((ps) => ps.color === chosenColor)
          const realCard = existingSet?.cards.find(
            (c) => "rentTable" in c && Array.isArray((c as any).rentTable) && (c as any).rentTable.length > 0
          ) as PropertyCard | undefined
          const fakeCard: PropertyCard = {
            id: card.id,
            name: card.name,
            color: chosenColor,
            value: card.value,
            rentTable: realCard?.rentTable ?? defaults.rentTable,
            setRequired: realCard?.setRequired ?? existingSet?.requiredCount ?? defaults.setRequired,
          }
          s.players[playerIdx].propertySets = addCardToPropertySets(
            s.players[playerIdx].propertySets,
            fakeCard
          )
          actions.push(`played ${card.name} on ${chosenColor}`)
        }
      }
    }

    // Bank money cards
    for (const card of toBank) {
      s.players[playerIdx].bankCards.push(card as never)
      s.players[playerIdx].bankTotal += card.value
      actions.push(`banked ${card.name}`)
    }

    // Update AI hand - remove played/banked cards, add drawn cards, trim to 7
    const playedIds = new Set([...toPlay, ...toBank].map((c) => c.id))
    const newHand = [...remaining, ...drawn.filter((c) => !playedIds.has(c.id))]
    s.aiHands[aiPlayerId] = newHand.slice(0, 7)
    s.deck = s.deck.slice(2)

    if (actions.length > 0) {
      s.moveLog.unshift(logEntry(aiPlayer.name, actions.join(", ")))
    }
  })

  // Check win for AI
  const winner = checkWin(get().players, aiPlayerId, get().setsToWin)
  if (winner) {
    set((s) => {
      s.winner = winner
      s.moveLog.unshift(logEntry(winner, "🏆 WON THE GAME!"))
    })
    return
  }

  // End AI turn
  get().rotateTurn()
},
