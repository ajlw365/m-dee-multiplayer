confirmStealProperty: (cardId, targetPlayerId, propertyId): ActionResult => {
  const state = get()
  const card = state.hand.find((c) => c.id === cardId)
  if (!card) return { ok: false, message: "Card not found." }
  const currentPlayer = state.players.find((p) => p.id === state.currentPlayerId)!
  const targetPlayer = state.players.find((p) => p.id === targetPlayerId)
  if (!targetPlayer) return { ok: false, message: "Target player not found." }

  set((s) => {
    s.hand = s.hand.filter((c) => c.id !== cardId)
    const currentIdx = s.players.findIndex((p) => p.id === s.currentPlayerId)
    const targetIdx = s.players.findIndex((p) => p.id === targetPlayerId)

    // Find and remove the property from target's sets
    let stolenCard: PropertyCard | undefined
    s.players[targetIdx].propertySets = s.players[targetIdx].propertySets
      .map((ps) => {
        const found = ps.cards.find((c) => c.id === propertyId)
        if (found) {
          stolenCard = found
          const remaining = ps.cards.filter((c) => c.id !== propertyId)
          return { ...ps, cards: remaining, isComplete: remaining.length >= ps.requiredCount }
        }
        return ps
      })
      .filter((ps) => ps.cards.length > 0)

    // Add stolen card to current player's sets
    if (stolenCard) {
      s.players[currentIdx].propertySets = addCardToPropertySets(
        s.players[currentIdx].propertySets,
        stolenCard
      )
    }

    s.actionsRemaining -= 1
    s.playAreaCard = card
    s.selectedCardId = null
    s.stealDialog = { open: false, mode: null, cardId: null, targetPlayerId: null }
    s.moveLog.unshift(logEntry(currentPlayer.name, `used Sly Deal to steal a property from ${targetPlayer.name}`))
  })

  if (get().actionsRemaining === 0) {
    setTimeout(() => get().endTurn(), 300)
  }

  const winner = checkWin(get().players, get().currentPlayerId, get().setsToWin)
  if (winner) {
    set((s) => {
      s.winner = winner
      s.moveLog.unshift(logEntry(winner, "🏆 WON THE GAME!"))
    })
  }

  return { ok: true, message: "Property stolen!" }
},
